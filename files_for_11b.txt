<?php

namespace App\Http\Controllers;

use App\Exceptions\Fintech\DuplicateWebhookException;
use App\Services\Financial\Mpesa\MpesaWebhookService;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Log;
use Throwable;

final class MpesaWebhookController extends Controller
{
    public function __construct(private readonly MpesaWebhookService $webhookService) {}

    /**
     * C2B Validation endpoint called by Safaricom prior to confirming Paybill payments.
     */
    public function c2bValidation(Request $request): JsonResponse
    {
        // Accept all payments by default; account reconciliation happens on confirmation
        return response()->json([
            'ResultCode' => 0,
            'ResultDesc' => 'Accepted',
        ]);
    }

    /**
     * C2B Confirmation endpoint called by Safaricom when cash is received.
     */
    public function c2bConfirmation(Request $request): JsonResponse
    {
        try {
            $payload = $request->all();
            $this->webhookService->ingestC2bConfirmation($payload);

            return response()->json([
                'ResultCode' => 0,
                'ResultDesc' => 'Confirmation received successfully',
            ]);
        } catch (DuplicateWebhookException $e) {
            Log::channel('daily')->info($e->getMessage());

            // Acknowledge duplicate safely to Safaricom so they stop retrying
            return response()->json([
                'ResultCode' => 0,
                'ResultDesc' => 'Duplicate acknowledged',
            ]);
        } catch (Throwable $e) {
            Log::channel('daily')->error('Error processing C2B Confirmation webhook: ' . $e->getMessage());

            // Still return 0 to Safaricom to prevent endless retry storms
            return response()->json([
                'ResultCode' => 0,
                'ResultDesc' => 'Ingested with internal logged error',
            ]);
        }
    }

    /**
     * STK Push Callback endpoint called by Safaricom after Lipa na M-Pesa prompt.
     */
    public function stkCallback(Request $request): JsonResponse
    {
        try {
            $payload = $request->all();
            $this->webhookService->ingestStkCallback($payload);

            return response()->json([
                'ResultCode' => 0,
                'ResultDesc' => 'STK callback processed',
            ]);
        } catch (DuplicateWebhookException $e) {
            Log::channel('daily')->info($e->getMessage());

            return response()->json([
                'ResultCode' => 0,
                'ResultDesc' => 'Duplicate STK acknowledged',
            ]);
        } catch (Throwable $e) {
            Log::channel('daily')->error('Error processing STK Callback: ' . $e->getMessage());

            return response()->json([
                'ResultCode' => 0,
                'ResultDesc' => 'Error logged internally',
            ]);
        }
    }

declare(strict_types=1);

namespace App\Services\Customer;

use App\Enums\RiskBand;
use App\Enums\UserRole;
use App\Exceptions\Fintech\DataProtectionConsentMissingException;
use App\Models\AuditTrail;
use App\Models\Borrower;
use App\Models\BorrowerFinancialProfile;
use App\Models\DataProtectionConsent;
use App\Models\User;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Log;
use Illuminate\Validation\ValidationException;
use Spatie\Permission\Models\Role;

final class BorrowerOnboardingService
{
    /**
     * Executes borrower registration, financial profiling, KDPA 2019 consent logging, login creation
     * (when a password is supplied) and audit tracking inside an atomic transaction.
     *
     * @param array $data Validated input data from BorrowerRegistrationRequest
     * @param string|null $ipAddress Client IP address
     * @param string|null $userAgent Client User Agent string
     * @return Borrower
     * @throws DataProtectionConsentMissingException
     * @throws ValidationException When the phone or email already belongs to a login account
     */
    public function register(array $data, ?string $ipAddress = null, ?string $userAgent = null): Borrower
    {
        $consentGranted = filter_var($data['data_protection_consent'] ?? false, FILTER_VALIDATE_BOOLEAN);

        if (! $consentGranted) {
            throw new DataProtectionConsentMissingException($data['id_number'] ?? 'UNKNOWN');
        }

        return DB::transaction(function () use ($data, $ipAddress, $userAgent) {
            $normalizedPhone = $this->normalizePhoneNumber((string) $data['phone_number']);

            $password = isset($data['password']) && $data['password'] !== ''
                ? (string) $data['password']
                : null;

            $email = isset($data['email']) && $data['email'] !== ''
                ? (string) $data['email']
                : null;

            $user = null;

            if ($password !== null) {
                $this->assertLoginDetailsAvailable($normalizedPhone, $email);

                /** @var User $user */
                $user = User::create([
                    'name' => $data['full_name'],
                    'email' => $email ?? $this->placeholderEmail($normalizedPhone),
                    'password' => $password, // hashed by the User model cast
                    'phone_number' => $normalizedPhone,
                    'is_active' => true,
                ]);

                Role::findOrCreate(UserRole::Borrower->value, 'web');
                $user->assignRole(UserRole::Borrower->value);
            }

            /** @var Borrower $borrower */
            $borrower = Borrower::create([
                'user_id' => $user?->id,
                'full_name' => $data['full_name'],
                'id_number' => $data['id_number'],
                'kra_pin' => isset($data['kra_pin']) ? strtoupper((string) $data['kra_pin']) : null,
                'phone_number' => $normalizedPhone,
                'email' => $data['email'] ?? null,
                'credit_limit' => '0.00',
                'risk_band' => RiskBand::C,
                'is_blacklisted' => false,
            ]);

            BorrowerFinancialProfile::create([
                'borrower_id' => $borrower->id,
                'monthly_income' => $data['monthly_income'],
                'monthly_expenses' => $data['monthly_expenses'],
                'mpesa_average_monthly_inflow' => $data['mpesa_average_monthly_inflow'] ?? '0.00',
                'existing_debt_obligations' => $data['existing_debt_obligations'] ?? '0.00',
            ]);

            DataProtectionConsent::create([
                'borrower_id' => $borrower->id,
                'consent_type' => 'CREDIT_SCORING_AND_M_PESA_DATA_PROCESSING',
                'is_granted' => true,
                'ip_address' => $ipAddress ?? '127.0.0.1',
                'user_agent' => $userAgent ?? 'SYSTEM',
                'granted_at' => now(),
            ]);

            AuditTrail::create([
                'user_id' => auth()->id(),
                'action' => 'BORROWER_ONBOARDED',
                'auditable_type' => Borrower::class,
                'auditable_id' => $borrower->id,
                'old_values' => null,
                'new_values' => [
                    'id_number' => $borrower->id_number,
                    'phone_number' => $borrower->phone_number,
                    'risk_band' => $borrower->risk_band->value,
                    'login_created' => $user !== null,
                ],
                'ip_address' => $ipAddress ?? '127.0.0.1',
                'user_agent' => $userAgent ?? 'SYSTEM',
            ]);

            Log::channel('fintech')->info("Borrower onboarding successful: ID {$borrower->id} [{$borrower->id_number}]");

            return $borrower;
        });
    }

    /**
     * Stops registration cleanly if the phone or email already belongs to a login account.
     *
     * @throws ValidationException
     */
    private function assertLoginDetailsAvailable(string $phone, ?string $email): void
    {
        $errors = [];

        if (User::query()->where('phone_number', $phone)->exists()) {
            $errors['phone_number'] = 'This phone number already has an account. Try signing in instead.';
        }

        if ($email !== null && User::query()->where('email', $email)->exists()) {
            $errors['email'] = 'This email address already has an account.';
        }

        if ($errors !== []) {
            throw ValidationException::withMessages($errors);
        }
    }

    /**
     * Borrowers without an email sign in by phone. The reserved .invalid ending can never be a real address.
     */
    private function placeholderEmail(string $normalizedPhone): string
    {
        return $normalizedPhone . '@borrowers.ethical-lending.invalid';
    }

    /**
     * Normalizes Kenyan phone numbers to E.164 Safaricom format (2547XXXXXXXX or 2541XXXXXXXX).
     */
    private function normalizePhoneNumber(string $phone): string
    {
        $cleaned = preg_replace('/\D/', '', $phone);

        if (str_starts_with($cleaned, '0')) {
            return '254'.substr($cleaned, 1);
        }

        if (str_starts_with($cleaned, '7') || str_starts_with($cleaned, '1')) {
            return '254'.$cleaned;
        }

        return $cleaned;
    }

<?php

declare(strict_types=1);

namespace App\Services\Governance;

use App\Models\AuditTrail;
use App\Models\LoanApplication;
use App\Models\MakerCheckerRequest;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Log;
use RuntimeException;

final class MakerCheckerService
{
    /**
     * Creates a pending dual-control authorization request.
     *
     * @param int $makerUserId
     * @string $actionType (e.g., 'APPROVE_LOAN_DISBURSEMENT', 'WAIVE_INTEREST')
     * @string $targetEntityType
     * @int $targetEntityId
     * @array $payload
     * @return MakerCheckerRequest
     */
    public function createPendingRequest(
        int $makerUserId,
        string $actionType,
        string $targetEntityType,
        int $targetEntityId,
        array $payload = []
    ): MakerCheckerRequest {
        return MakerCheckerRequest::create([
            'maker_user_id' => $makerUserId,
            'action_type' => $actionType,
            'target_entity_type' => $targetEntityType,
            'target_entity_id' => $targetEntityId,
            'payload' => $payload,
            'status' => 'pending',
        ]);
    }

    /**
     * Authorizes and executes a pending request under strict non-self-approval enforcement.
     *
     * @param int $requestId
     * @param int $checkerUserId
     * @param string|null $checkerNotes
     * @return array
     * @throws RuntimeException
     */
    public function authorizeRequest(int $requestId, int $checkerUserId, ?string $checkerNotes = null): array
    {
        return DB::transaction(function () use ($requestId, $checkerUserId, $checkerNotes) {
            /** @var MakerCheckerRequest $request */
            $request = MakerCheckerRequest::where('id', $requestId)
                ->lockForUpdate()
                ->firstOrFail();

            if ($request->status !== 'pending') {
                throw new RuntimeException("Maker-Checker Request #{$requestId} is not in pending status.");
            }

            // Dual-Control Requirement: Checker MUST NOT be the Maker.
            // Compared as integers so a driver that returns ids as strings cannot slip past the check.
            if ((int) $request->maker_user_id === $checkerUserId) {
                Log::channel('fintech')->warning("Dual-Control Violation Attempt: User #{$checkerUserId} attempted self-approval on Request #{$requestId}");
                throw new RuntimeException("Dual-Control Violation: You cannot authorize a request you initiated.");
            }

            $request->update([
                'status' => 'approved',
                'checker_user_id' => $checkerUserId,
                'checker_notes' => $checkerNotes,
                'actioned_at' => now(),
            ]);

            // Execute specific state mutation based on action type
            if ($request->action_type === 'APPROVE_LOAN_APPLICATION') {
                /** @var LoanApplication|null $app */
                $app = LoanApplication::where('id', $request->target_entity_id)->first();
                if ($app) {
                    $app->update(['status' => 'approved']);
                }
            }

            // Record immutable Audit Trail
            AuditTrail::create([
                'user_id' => $checkerUserId,
                'action' => 'MAKER_CHECKER_AUTHORIZED',
                'auditable_type' => MakerCheckerRequest::class,
                'auditable_id' => $request->id,
                'old_values' => ['status' => 'pending'],
                'new_values' => [
                    'status' => 'approved',
                    'action_type' => $request->action_type,
                    'checker_user_id' => $checkerUserId,
                ],
                'ip_address' => request()->ip() ?? '127.0.0.1',
                'user_agent' => request()->userAgent() ?? 'SYSTEM',
            ]);

            Log::channel('fintech')->info("Maker-Checker Request #{$requestId} [{$request->action_type}] authorized by Checker #{$checkerUserId}");

            return [
                'request_id' => $request->id,
                'action_type' => $request->action_type,
                'status' => 'approved',
            ];
        });
    }

    /**
     * Rejects a pending dual-control request and records the decision on the audit trail.
     */
    public function rejectRequest(int $requestId, int $checkerUserId, string $reason): array
    {
        return DB::transaction(function () use ($requestId, $checkerUserId, $reason) {
            /** @var MakerCheckerRequest $request */
            $request = MakerCheckerRequest::where('id', $requestId)
                ->lockForUpdate()
                ->firstOrFail();

            if ($request->status !== 'pending') {
                throw new RuntimeException("Maker-Checker Request #{$requestId} is not in pending status.");
            }

            if ((int) $request->maker_user_id === $checkerUserId) {
                Log::channel('fintech')->warning("Dual-Control Violation Attempt: User #{$checkerUserId} attempted to reject own Request #{$requestId}");
                throw new RuntimeException("Dual-Control Violation: You cannot reject/action a request you initiated.");
            }

            $request->update([
                'status' => 'rejected',
                'checker_user_id' => $checkerUserId,
                'checker_notes' => $reason,
                'actioned_at' => now(),
            ]);

            AuditTrail::create([
                'user_id' => $checkerUserId,
                'action' => 'MAKER_CHECKER_REJECTED',
                'auditable_type' => MakerCheckerRequest::class,
                'auditable_id' => $request->id,
                'old_values' => ['status' => 'pending'],
                'new_values' => [
                    'status' => 'rejected',
                    'action_type' => $request->action_type,
                    'checker_user_id' => $checkerUserId,
                    'reason' => $reason,
                ],
                'ip_address' => request()->ip() ?? '127.0.0.1',
                'user_agent' => request()->userAgent() ?? 'SYSTEM',
            ]);

            Log::channel('fintech')->info("Maker-Checker Request #{$requestId} [{$request->action_type}] rejected by Checker #{$checkerUserId}");

            return [
                'request_id' => $request->id,
                'status' => 'rejected',
            ];
        });
    }

<?php

declare(strict_types=1);

namespace App\Services\Customer;

use App\Enums\LoanStatus;
use App\Models\Borrower;
use App\Models\Loan;
use App\Models\LoanLedgerEntry;
use App\Services\Servicing\RestructuringEngine;
use Carbon\CarbonImmutable;
use Illuminate\Support\Facades\DB;

final class BorrowerPortalService
{
    /** Extra days of time a borrower can choose from. */
    public const EXTENSION_OPTIONS = [14, 21, 30, 45, 60];

    public const DEFAULT_EXTENSION_DAYS = 30;

    /** Mirrors the threshold inside RestructuringEngine. The engine remains the final judge. */
    private const MINIMUM_ELIGIBILITY_RATIO = '0.30';

    public function __construct(private readonly RestructuringEngine $engine)
    {
    }

    /**
     * Everything the borrower dashboard needs.
     *
     * @return array<string, mixed>
     */
    public function summary(Borrower $borrower): array
    {
        $loan = $this->currentLoan($borrower);

        if ($loan === null) {
            return ['loan' => null];
        }

        $balances = $this->balances($loan);
        $owed = $this->owed($balances);
        $due = $loan->due_date !== null ? CarbonImmutable::parse($loan->due_date) : null;

        return [
            'loan'       => $loan,
            'balances'   => $balances,
            'owed'       => $owed,
            'due_date'   => $due?->format('j M Y'),
            'is_overdue' => $due !== null && $due->endOfDay()->isPast() && bccomp($owed, '0.00', 2) > 0,
            'meter'      => $this->inDuplumMeter($loan),
            'relief'     => $this->canRequestRelief($loan) ? $this->reliefPreview($borrower, $balances) : null,
        ];
    }

    /** The borrower's latest loan that is not settled or written off. */
    public function currentLoan(Borrower $borrower): ?Loan
    {
        /** @var Loan|null $loan */
        $loan = $borrower->loans()
            ->whereNotIn('status', [LoanStatus::SETTLED->value, LoanStatus::WRITTEN_OFF->value])
            ->with('inDuplumTracking')
            ->latest('id')
            ->first();

        return $loan;
    }

    /**
     * Latest balances from the append-only ledger.
     *
     * @return array{principal: string, interest: string, fees: string}
     */
    public function balances(Loan $loan): array
    {
        $latest = LoanLedgerEntry::query()->where('loan_id', $loan->id)->latest('id')->first();

        if ($latest === null) {
            return [
                'principal' => (string) $loan->principal_disbursed,
                'interest'  => '0.00',
                'fees'      => '0.00',
            ];
        }

        return [
            'principal' => (string) $latest->running_principal_balance,
            'interest'  => (string) $latest->running_interest_balance,
            'fees'      => (string) $latest->running_fees_balance,
        ];
    }

    /**
     * Data for the Section 44A safety meter.
     * States: not_triggered (no default yet), accruing (charges counting), frozen (cap reached).
     *
     * @return array{state: string, accrued: string, cap: string, percent: string, headroom: string}
     */
    public function inDuplumMeter(Loan $loan): array
    {
        $tracking = $loan->inDuplumTracking;

        if ($tracking === null) {
            return [
                'state'    => 'not_triggered',
                'accrued'  => '0.00',
                'cap'      => '0.00',
                'percent'  => '0.00',
                'headroom' => '0.00',
            ];
        }

        $accrued = bcadd(
            (string) $tracking->total_post_default_interest,
            (string) $tracking->total_post_default_penalties,
            2
        );
        $cap = (string) $tracking->statutory_cap_limit;

        $percent = '0.00';
        if (bccomp($cap, '0.00', 2) > 0) {
            $percent = bcmul(bcdiv($accrued, $cap, 4), '100', 2);
            if (bccomp($percent, '100.00', 2) > 0) {
                $percent = '100.00';
            }
        }

        $headroom = bcsub($cap, $accrued, 2);
        if (bccomp($headroom, '0.00', 2) < 0) {
            $headroom = '0.00';
        }

        $frozen = (bool) $tracking->is_frozen || (bool) $loan->is_in_duplum_locked;

        return [
            'state'    => $frozen ? 'frozen' : 'accruing',
            'accrued'  => $accrued,
            'cap'      => $cap,
            'percent'  => $percent,
            'headroom' => $headroom,
        ];
    }

    /** Relief is offered on defaulted loans, or disbursed loans past their due date. */
    public function canRequestRelief(Loan $loan): bool
    {
        if ($loan->status === LoanStatus::DEFAULTED) {
            return true;
        }

        return $loan->status === LoanStatus::DISBURSED
            && $loan->due_date !== null
            && CarbonImmutable::parse($loan->due_date)->endOfDay()->isPast();
    }

    /**
     * A read-only preview that follows the same rules as RestructuringEngine
     * (penalties forgiven, interest kept, balance split into two instalments).
     * Nothing is saved here.
     *
     * @param  array{principal: string, interest: string, fees: string}  $balances
     * @return array<string, mixed>
     */
    public function reliefPreview(Borrower $borrower, array $balances): array
    {
        $ratio = $this->engine->calculateRepaymentRatio($borrower);

        $owedToday = $this->owed($balances);
        $owedAfter = bcadd($balances['principal'], $balances['interest'], 2);

        $first = bcdiv($owedAfter, '2.00', 2);
        $remainder = bcsub($owedAfter, bcmul($first, '2.00', 2), 2);
        $second = bcadd($first, $remainder, 2);

        $today = CarbonImmutable::now();
        $schedule = [];
        foreach (self::EXTENSION_OPTIONS as $days) {
            $schedule[$days] = [
                'first_due'  => $today->addDays(intdiv($days, 2))->format('j M Y'),
                'second_due' => $today->addDays($days)->format('j M Y'),
            ];
        }

        return [
            'eligible'     => bccomp($ratio, self::MINIMUM_ELIGIBILITY_RATIO, 2) >= 0,
            'owed_today'   => $owedToday,
            'waived'       => $balances['fees'],
            'owed_after'   => $owedAfter,
            'first_amount' => $first,
            'second_amount' => $second,
            'options'      => self::EXTENSION_OPTIONS,
            'default_days' => self::DEFAULT_EXTENSION_DAYS,
            'schedule'     => $schedule,
        ];
    }

    /**
     * Creates and accepts a restructuring proposal in one transaction,
     * using the same engine the lender side relies on.
     */
    public function acceptRelief(Loan $loan, int $extensionDays): Loan
    {
        if (! in_array($extensionDays, self::EXTENSION_OPTIONS, true)) {
            throw new \InvalidArgumentException('Unsupported extension period.');
        }

        if (! $this->canRequestRelief($loan)) {
            throw new \DomainException('This loan is not in arrears, so relief cannot be requested.');
        }

        return DB::transaction(function () use ($loan, $extensionDays): Loan {
            $proposal = $this->engine->generateProposal($loan, $extensionDays);

            return $this->engine->acceptProposal($proposal);
        });
    }

    /** @param array{principal: string, interest: string, fees: string} $balances */
    private function owed(array $balances): string
    {
        return bcadd(bcadd($balances['principal'], $balances['interest'], 2), $balances['fees'], 2);
    }
