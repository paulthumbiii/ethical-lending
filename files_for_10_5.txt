it('allocates partial repayment to fees first then interest, leaving principal intact', function (): void {
    // Borrower pays KES 1,000.00
    // Should clear KES 500 fees entirely, remaining KES 500 to interest, KES 0 to principal
    $result = $this->service->executeWaterfall($this->loan, '1000.00', 'MPESA-TXN-01');

    expect($result['allocated_fees'])->toBe('500.00')
        ->and($result['allocated_interest'])->toBe('500.00')
        ->and($result['allocated_principal'])->toBe('0.00')
        ->and($result['overpayment_credit'])->toBe('0.00')
        ->and($result['new_total_balance'])->toBe('11000.00')
        ->and($result['is_settled'])->toBeFalse();

    // Verify ledger entry
    $latest = LoanLedgerEntry::where('loan_id', $this->loan->id)->latest('id')->first();
    expect($latest->running_fees_balance)->toBe('0.00')
        ->and($latest->running_interest_balance)->toBe('1000.00')
        ->and($latest->running_principal_balance)->toBe('10000.00');
});

it('cascades sequential payments through interest down into principal', function (): void {
    // First payment: KES 1,000 (clears 500 fee, 500 interest)
    $this->service->executeWaterfall($this->loan, '1000.00', 'MPESA-TXN-01');

    // Second payment: KES 3,000
    // Fees are 0. Remaining interest is KES 1,000 (cleared). Remaining KES 2,000 goes to principal!
    $result = $this->service->executeWaterfall($this->loan, '3000.00', 'MPESA-TXN-02');

    expect($result['allocated_fees'])->toBe('0.00')
        ->and($result['allocated_interest'])->toBe('1000.00')
        ->and($result['allocated_principal'])->toBe('2000.00')
        ->and($result['new_total_balance'])->toBe('8000.00')
        ->and($result['is_settled'])->toBeFalse();

    $latest = LoanLedgerEntry::where('loan_id', $this->loan->id)->latest('id')->first();
    expect($latest->running_fees_balance)->toBe('0.00')
        ->and($latest->running_interest_balance)->toBe('0.00')
        ->and($latest->running_principal_balance)->toBe('8000.00');
});

it('fully settles loan and emits an outbox event when entire debt is cleared', function (): void {
    // Pay exact total balance of KES 12,000.00
    $result = $this->service->executeWaterfall($this->loan, '12000.00', 'MPESA-FULL-SETTLE');

    expect($result['allocated_fees'])->toBe('500.00')
        ->and($result['allocated_interest'])->toBe('1500.00')
        ->and($result['allocated_principal'])->toBe('10000.00')
        ->and($result['new_total_balance'])->toBe('0.00')
        ->and($result['is_settled'])->toBeTrue();

    $this->loan->refresh();
    expect($this->loan->status)->toBe(LoanStatus::SETTLED);

    // Verify outbox settlement event
    $outbox = OutboxEvent::where('aggregate_id', $this->loan->id)
        ->where('event_name', 'loan.settled')
        ->first();

        ->and($outbox->payload['status'])->toBe('SETTLED');
});

    /**
     * Executes the statutory payment waterfall under a pessimistic row lock.
     * Order of allocation: Fees & Penalties -> Accrued Interest -> Principal.
     *
     * @return array{
     *     allocated_fees: string,
     *     allocated_interest: string,
     *     allocated_principal: string,
     *     overpayment_credit: string,
     *     new_total_balance: string,
     *     is_settled: bool
     * }
     */
    public function executeWaterfall(
        Loan $loan,
        string $amount,
        string $reference,
        string $source = 'MPESA_C2B'
    ): array {
        // Step 1: Validate numeric positive amount
        if (!is_numeric($amount) || bccomp($amount, '0.00', 2) <= 0) {
            throw new InvalidPaymentAmountException($amount);
        }

        return DB::transaction(function () use ($loan, $amount, $reference, $source): array {
            // Step 2: Pessimistic Row Lock on Loan
            $lockedLoan = Loan::query()->where('id', $loan->id)->lockForUpdate()->firstOrFail();

            // Step 3: Retrieve current running balances from the latest immutable ledger line
            $latestEntry = LoanLedgerEntry::query()
                ->where('loan_id', $lockedLoan->id)
                ->latest('id')
                ->lockForUpdate()
                ->first();

            $currentFees = $latestEntry !== null ? (string) $latestEntry->running_fees_balance : '0.00';
            $currentInterest = $latestEntry !== null ? (string) $latestEntry->running_interest_balance : '0.00';
            $currentPrincipal = $latestEntry !== null ? (string) $latestEntry->running_principal_balance : (string) $lockedLoan->principal_disbursed;

            $unallocated = $amount;

            // Step 4: Allocate to Fees & Penalties first
            $allocatedFees = '0.00';
            if (bccomp($currentFees, '0.00', 2) > 0) {
                $allocatedFees = bccomp($unallocated, $currentFees, 2) >= 0 ? $currentFees : $unallocated;
                $unallocated = bcsub($unallocated, $allocatedFees, 2);
            }

            // Step 5: Allocate to Accrued Interest second
            $allocatedInterest = '0.00';
            if (bccomp($unallocated, '0.00', 2) > 0 && bccomp($currentInterest, '0.00', 2) > 0) {
                $allocatedInterest = bccomp($unallocated, $currentInterest, 2) >= 0 ? $currentInterest : $unallocated;
                $unallocated = bcsub($unallocated, $allocatedInterest, 2);
            }

            // Step 6: Allocate to Outstanding Principal third
            $allocatedPrincipal = '0.00';
            if (bccomp($unallocated, '0.00', 2) > 0 && bccomp($currentPrincipal, '0.00', 2) > 0) {
                $allocatedPrincipal = bccomp($unallocated, $currentPrincipal, 2) >= 0 ? $currentPrincipal : $unallocated;
                $unallocated = bcsub($unallocated, $allocatedPrincipal, 2);
            }

            // Step 7: Any remaining unallocated cash becomes Overpayment Credit
            $overpaymentCredit = $unallocated;

            // Step 8: Compute new running balances
            $newFees = bcsub($currentFees, $allocatedFees, 2);
            $newInterest = bcsub($currentInterest, $allocatedInterest, 2);
            $newPrincipal = bcsub($currentPrincipal, $allocatedPrincipal, 2);
            $newTotalBalance = bcadd(bcadd($newPrincipal, $newInterest, 2), $newFees, 2);

            // Total credit applied against the loan
            $totalCreditApplied = bcadd(bcadd($allocatedFees, $allocatedInterest, 2), $allocatedPrincipal, 2);

            // Step 9: Write immutable append-only ledger entry
            LoanLedgerEntry::create([
                'loan_id'                   => $lockedLoan->id,
                'entry_type'                => 'repayment',
                'debit'                     => '0.00',
                'credit'                    => $totalCreditApplied,
                'running_principal_balance' => $newPrincipal,
                'running_interest_balance'  => $newInterest,
                'running_fees_balance'      => $newFees,
                'total_running_balance'     => $newTotalBalance,
                'reference'                 => $reference,
                'notes'                     => sprintf(
                    'Repayment via %s [Ref: %s]. Waterfall split: Fees: KES %s | Interest: KES %s | Principal: KES %s | Overpayment: KES %s',
                    $source,
                    $reference,
                    $allocatedFees,
                    $allocatedInterest,
                    $allocatedPrincipal,
                    $overpaymentCredit
                ),
            ]);

            // Step 10: Log waterfall breakdown
            DB::table('payment_waterfall_logs')->insert([
                'loan_id'                 => $lockedLoan->id,
                'payment_reference'       => $reference,
                'total_amount_received'   => $amount,
                'allocated_to_fees'       => $allocatedFees,
                'allocated_to_interest'   => $allocatedInterest,
                'allocated_to_principal'  => $allocatedPrincipal,
                'overpayment_amount'      => $overpaymentCredit,
                'created_at'              => CarbonImmutable::now(),
            ]);

            // Step 11: Check for full loan settlement
            $isSettled = bccomp($newTotalBalance, '0.00', 2) === 0;
            if ($isSettled) {
                $lockedLoan->update([
                    'status' => LoanStatus::SETTLED,
                ]);

                // Cryptographically chained Outbox Event for settlement
                $prevEvent = OutboxEvent::query()->latest('id')->first();
                $prevHash = $prevEvent ? $prevEvent->current_event_hash : str_repeat('0', 64);

                $payload = [
                    'loan_id'          => $lockedLoan->id,
                    'loan_number'      => $lockedLoan->loan_number,
                    'borrower_id'      => $lockedLoan->borrower_id,
                    'status'           => 'SETTLED',
                    'final_payment_ref'=> $reference,
                    'settled_at'       => CarbonImmutable::now()->toIso8601String(),
                ];

                $currentHash = hash('sha256', $prevHash . 'loan.settled' . json_encode($payload));

                OutboxEvent::create([
                    'aggregate_type'      => Loan::class,
                    'aggregate_id'        => $lockedLoan->id,
                    'event_name'          => 'loan.settled',
                    'payload'             => $payload,
                    'previous_event_hash' => $prevHash,
                    'current_event_hash'  => $currentHash,
                    'is_published'        => false,
                    'retry_count'         => 0,
                ]);
            }

            // Step 12: Record operational audit trail
            AuditTrail::create([
                'user_id'        => auth()->id(),
                'action'         => 'PAYMENT_WATERFALL_EXECUTED',
                'auditable_type' => Loan::class,
                'auditable_id'   => $lockedLoan->id,

                    'interest'  => $currentInterest,
                    'fees'      => $currentFees,
                ],

                    'principal'   => $newPrincipal,
                    'interest'    => $newInterest,
                    'fees'        => $newFees,
                    'overpayment' => $overpaymentCredit,
                    'is_settled'  => $isSettled,
                ],
                'ip_address'     => request()->ip() ?? '127.0.0.1',
                'user_agent'     => request()->userAgent() ?? 'System/CLI',
            ]);

            Log::channel('daily')->info(sprintf(
                'Waterfall executed on Loan #%d: Received KES %s -> Fees: %s, Int: %s, Prin: %s, Overpay: %s [Ref: %s]',
                $lockedLoan->id,
                $amount,
                $allocatedFees,
                $allocatedInterest,
                $allocatedPrincipal,
                $overpaymentCredit,
                $reference
            ));

            return [
                'allocated_fees'      => $allocatedFees,
                'allocated_interest'  => $allocatedInterest,
                'allocated_principal' => $allocatedPrincipal,
                'overpayment_credit'  => $overpaymentCredit,
                'new_total_balance'   => $newTotalBalance,
                'is_settled'          => $isSettled,
            ];
        });
    }

    /**
     * Ingests incoming C2B Paybill Confirmation webhook from Safaricom.
     *
     * @param array<string, mixed> $payload
     */
    public function ingestC2bConfirmation(array $payload): MpesaC2bCallback
    {
        $transId = (string) ($payload['TransID'] ?? '');
        $amount = (string) ($payload['TransAmount'] ?? '0.00');
        $billRef = trim((string) ($payload['BillRefNumber'] ?? ''));
        $msisdn = (string) ($payload['MSISDN'] ?? '');

        if ($transId === '') {
            throw new \InvalidArgumentException('TransID missing from Safaricom C2B payload.');
        }

        // Idempotency Check: Prevent duplicate webhook processing
        $existing = MpesaC2bCallback::query()->where('trans_id', $transId)->first();
        if ($existing !== null) {
            throw new DuplicateWebhookException($transId, 'SAFARICOM_C2B');
        }

        $callback = MpesaC2bCallback::create([
            'transaction_type'    => (string) ($payload['TransactionType'] ?? 'Pay Utility'),
            'trans_id'            => $transId,
            'trans_time'          => (string) ($payload['TransTime'] ?? CarbonImmutable::now()->format('YmdHis')),
            'trans_amount'        => $amount,
            'business_short_code' => (string) ($payload['BusinessShortCode'] ?? config('mpesa.shortcode')),
            'bill_ref_number'     => $billRef,
            'msisdn'              => $msisdn,
            'first_name'          => (string) ($payload['FirstName'] ?? null),
            'is_reconciled'       => false,
        ]);

        // Dispatch background job for high-concurrency ledger crediting
        ProcessMpesaCallbackJob::dispatch('C2B', $callback->id);

        return $callback;
    }

    /**
     * Ingests incoming Lipa na M-Pesa STK Push Callback from Safaricom.
     *
     * @param array<string, mixed> $payload
     */
    public function ingestStkCallback(array $payload): MpesaStkRequest
    {
        $stkCallback = $payload['Body']['stkCallback'] ?? [];
        $checkoutRequestId = (string) ($stkCallback['CheckoutRequestID'] ?? '');
        $resultCode = (int) ($stkCallback['ResultCode'] ?? -1);
        $resultDesc = (string) ($stkCallback['ResultDesc'] ?? 'No description');

        $stkRequest = MpesaStkRequest::query()
            ->where('checkout_request_id', $checkoutRequestId)
            ->firstOrFail();

        // Check if already processed
        if ($stkRequest->status === 'successful') {
            throw new DuplicateWebhookException($checkoutRequestId, 'SAFARICOM_STK');
        }

        $metadataItems = $stkCallback['CallbackMetadata']['Item'] ?? [];
        $receiptNumber = null;
        $amount = (string) $stkRequest->amount;

        foreach ($metadataItems as $item) {
            if (($item['Name'] ?? '') === 'MpesaReceiptNumber') {
                $receiptNumber = (string) $item['Value'];
            }
            if (($item['Name'] ?? '') === 'Amount') {
                $amount = number_format((float) $item['Value'], 2, '.', '');
            }
        }

        if ($resultCode === 0 && $receiptNumber !== null) {
            $stkRequest->update([
                'status'               => 'successful',
                'mpesa_receipt_number' => $receiptNumber,
                'result_code'          => $resultCode,
                'result_desc'          => $resultDesc,
                'amount'               => $amount,
            ]);

            // Dispatch background job for waterfall execution
            ProcessMpesaCallbackJob::dispatch('STK', $stkRequest->id);
        } else {
            $stkRequest->update([
                'status'      => 'failed',
                'result_code' => $resultCode,
                'result_desc' => $resultDesc,
            ]);
        }

        return $stkRequest;
    }

    /**
     * Asynchronously executes payment waterfall and double-entry postings for an ingested callback.
     */
    public function processRepayment(string $sourceType, int $recordId): void
    {
        DB::transaction(function () use ($sourceType, $recordId): void {
            if ($sourceType === 'C2B') {
                $callback = MpesaC2bCallback::query()->where('id', $recordId)->lockForUpdate()->firstOrFail();
                if ($callback->is_reconciled) {
                    return; // Already reconciled
                }

                $loan = $this->resolveLoanFromReference($callback->bill_ref_number, $callback->msisdn);
                if ($loan === null) {
                    Log::channel('daily')->warning(sprintf(
                        'Unmatched C2B Payment: No active loan found for BillRef [%s] or MSISDN [%s] [TransID: %s]',
                        $callback->bill_ref_number,
                        $callback->msisdn,
                        $callback->trans_id
                    ));
                    return;
                }

                $amount = (string) $callback->trans_amount;
                $reference = (string) $callback->trans_id;

                // 1. Execute Legal Payment Waterfall
                $waterfall = $this->waterfallService->executeWaterfall(
                    loan: $loan,
                    amount: $amount,
                    reference: $reference,
                    source: 'MPESA_PAYBILL_C2B'
                );

                // 2. Post Balanced Double-Entry General Ledger Entry
                $this->glService->recordRepaymentPosting(
                    loan: $loan,
                    feesPaid: $waterfall['allocated_fees'],
                    interestPaid: $waterfall['allocated_interest'],
                    principalPaid: $waterfall['allocated_principal'],
                    overpayment: $waterfall['overpayment_credit'],
                    reference: $reference
                );

                $callback->update(['is_reconciled' => true]);
            } elseif ($sourceType === 'STK') {
                $stk = MpesaStkRequest::query()->where('id', $recordId)->lockForUpdate()->firstOrFail();
                if ($stk->status !== 'successful' || $stk->loan_id === null) {
                    return;
                }

                $loan = Loan::query()->where('id', $stk->loan_id)->firstOrFail();
                $amount = (string) $stk->amount;
                $reference = (string) $stk->mpesa_receipt_number;

                // 1. Execute Legal Payment Waterfall
                $waterfall = $this->waterfallService->executeWaterfall(
                    loan: $loan,
                    amount: $amount,
                    reference: $reference,
                    source: 'MPESA_STK_PUSH'
                );

                // 2. Post Balanced Double-Entry General Ledger Entry
                $this->glService->recordRepaymentPosting(
                    loan: $loan,
                    feesPaid: $waterfall['allocated_fees'],
                    interestPaid: $waterfall['allocated_interest'],
                    principalPaid: $waterfall['allocated_principal'],
                    overpayment: $waterfall['overpayment_credit'],
                    reference: $reference
                );
            }
        });
    }

    /**
     * Resolves active loan by Loan Number, Application Number, or borrower's registered phone.
     */
    private function resolveLoanFromReference(string $reference, string $phone): ?Loan
    {
        // 1. Match exact loan number (e.g. LN-B2C-01)
        $loan = Loan::query()->where('loan_number', $reference)->first();
        if ($loan !== null) {
            return $loan;
        }

        // 2. Match active borrower loan by phone
        return Loan::query()
            ->whereHas('borrower', fn ($q) => $q->where('phone_number', $phone))
            ->whereIn('status', ['disbursed', 'defaulted', 'watch'])
            ->latest('id')
            ->first();
    }
