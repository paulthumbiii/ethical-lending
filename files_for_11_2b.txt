enum UserRole: string
{
    case LenderAdmin = 'lender_admin';
    case CreditOfficer = 'credit_officer';
    case CollectionsOfficer = 'collections_officer';
    case FinanceOfficer = 'finance_officer';
    case Borrower = 'borrower';

    public static function staff(): array
    {
        return [
            self::LenderAdmin->value,
            self::CreditOfficer->value,
            self::CollectionsOfficer->value,
            self::FinanceOfficer->value,
        ];
    }

    /** Pipe-separated list, the format Spatie's role middleware expects. */
    public static function staffPipe(): string
    {
        return implode('|', self::staff());

        return implode('|', self::approvers());

    public static function collectionsViewers(): array

        ];

<?php

declare(strict_types=1);

namespace App\Services\Underwriting;

use App\Enums\RiskBand;
use App\Models\Borrower;
use App\Models\BorrowerFinancialProfile;
use App\Models\CrbInquiryRecord;
use App\Models\CreditRuleEvaluation;
use App\Models\LoanApplication;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Log;

final class CreditUnderwritingEngine
{
    /**
     * Evaluates a loan application against statutory and risk underwriting rules.
     * Assigns risk bands, calculates credit limits, and records detailed rule logs.
     *
     * @param int $applicationId
     * @return array Summary of underwriting decision and assigned terms
     */
    public function evaluateApplication(int $applicationId): array
    {
        return DB::transaction(function () use ($applicationId) {
            /** @var LoanApplication $application */
            $application = LoanApplication::where('id', $applicationId)
                ->lockForUpdate()
                ->firstOrFail();

            /** @var Borrower $borrower */
            $borrower = Borrower::where('id', $application->borrower_id)
                ->lockForUpdate()
                ->firstOrFail();

            /** @var BorrowerFinancialProfile|null $profile */
            $profile = BorrowerFinancialProfile::where('borrower_id', $borrower->id)->first();

            /** @var CrbInquiryRecord|null $latestCrb */
            $latestCrb = CrbInquiryRecord::where('borrower_id', $borrower->id)
                ->orderByDesc('id')
                ->first();

            $passedCount = 0;
            $failedCount = 0;

            // 1. Hard Rule: Borrower Blacklist Inspection
            $isBlacklisted = $borrower->is_blacklisted;
            $this->recordRuleEvaluation(
                $application->id,
                'RULE_01_BLACKLIST_CHECK',
                'Borrower Fraud & Blacklist Inspection',
                ! $isBlacklisted ? 'PASSED' : 'FAILED',
                $isBlacklisted ? 'Blacklisted' : 'Clean',
                'Clean'
            );
            $isBlacklisted ? $failedCount++ : $passedCount++;

            // 2. Hard Rule: CRB Defaulted Accounts Audit
            $defaultedAccounts = $latestCrb ? $latestCrb->defaulted_accounts_count : 0;
            $crbPassed = $defaultedAccounts === 0;
            $this->recordRuleEvaluation(
                $application->id,
                'RULE_02_CRB_DEFAULTS',
                'CRB Active NPL/Defaulted Accounts Check',
                $crbPassed ? 'PASSED' : 'FAILED',
                (string) $defaultedAccounts,
                '0'
            );
            $crbPassed ? $passedCount++ : $failedCount++;

            // 3. Hard Rule: Debt-To-Income (DTI) Ratio Guard (Max 50.00%)
            $monthlyIncome = $profile ? (string) $profile->monthly_income : '0.00';
            $monthlyExpenses = $profile ? (string) $profile->monthly_expenses : '0.00';
            $existingDebts = $profile ? (string) $profile->existing_debt_obligations : '0.00';

            $totalObligations = bcadd($monthlyExpenses, $existingDebts, 2);
            $dtiRatio = bccomp($monthlyIncome, '0.00', 2) > 0
                ? bcdiv($totalObligations, $monthlyIncome, 4)
                : '1.0000';

            $dtiPassed = bccomp($dtiRatio, '0.5000', 4) <= 0;
            $dtiPercentage = bcmul($dtiRatio, '100', 2).'%';

            $this->recordRuleEvaluation(
                $application->id,
                'RULE_03_DTI_RATIO',
                'Debt-To-Income Ratio Below 50%',
                $dtiPassed ? 'PASSED' : 'FAILED',
                $dtiPercentage,
                '<= 50.00%'
            );
            $dtiPassed ? $passedCount++ : $failedCount++;

            // 4. Hard Rule: M-Pesa Cash Flow Capacity Test (Min 1.5x Requested Principal)
            $mpesaInflow = $profile ? (string) $profile->mpesa_average_monthly_inflow : '0.00';
            $requestedPrincipal = (string) $application->requested_principal;

            $requiredInflow = bcmul($requestedPrincipal, '1.50', 2);
            $inflowPassed = bccomp($mpesaInflow, $requiredInflow, 2) >= 0;

            $this->recordRuleEvaluation(
                $application->id,
                'RULE_04_MPESA_INFLOW_CAPACITY',
                'M-Pesa Average Monthly Inflow Capacity',
                $inflowPassed ? 'PASSED' : 'FAILED',
                "KES {$mpesaInflow}",
                ">= KES {$requiredInflow}"
            );
            $inflowPassed ? $passedCount++ : $failedCount++;

            // Decision Matrix & Term Assignment
            if ($failedCount > 0) {
                $status = 'rejected';
                $riskBand = RiskBand::D;
                $approvedLimit = '0.00';
                $rejectionReason = "Failed {$failedCount} underwriting risk evaluation rules.";
            } else {
                $status = 'approved_pending_checker';
                $rejectionReason = null;

                // Score-Based Risk Band Assignment
                $crbScore = $latestCrb ? $latestCrb->credit_score : 600;

                if ($crbScore >= 750) {
                    $riskBand = RiskBand::A;
                    $approvedLimit = bcmul($requestedPrincipal, '1.25', 2);
                } elseif ($crbScore >= 650) {
                    $riskBand = RiskBand::B;
                    $approvedLimit = $requestedPrincipal;
                } else {
                    $riskBand = RiskBand::C;
                    $approvedLimit = bcmul($requestedPrincipal, '0.75', 2);
                }
            }

            // Update Application state
            $application->update([
                'status' => $status,
                'rejection_reason' => $rejectionReason,
            ]);

            // Update Borrower risk profile and credit limit
            $borrower->update([
                'risk_band' => $riskBand,
                'credit_limit' => $approvedLimit,
            ]);

            Log::channel('fintech')->info("Underwriting complete for Application #{$application->id}: Status = {$status}, Risk Band = {$riskBand->value}, Approved Limit = KES {$approvedLimit}");

            return [
                'application_id' => $application->id,
                'status' => $status,
                'risk_band' => $riskBand->value,
                'approved_limit' => $approvedLimit,
                'rules_passed' => $passedCount,
                'rules_failed' => $failedCount,
            ];
        });
    }

    private function recordRuleEvaluation(
        int $applicationId,
        string $identifier,
        string $description,
        string $status,
        string $actualValue,
        string $thresholdValue
    ): void {
        CreditRuleEvaluation::create([
            'loan_application_id' => $applicationId,
            'rule_identifier' => $identifier,
            'rule_description' => $description,
            'status' => $status,
            'actual_value' => $actualValue,
            'threshold_value' => $thresholdValue,
        ]);
    }

declare(strict_types=1);

namespace App\Http\Controllers\Borrower;

use App\Http\Controllers\Controller;
use App\Http\Requests\StoreLoanApplicationRequest;
use App\Models\AuditTrail;
use App\Models\Borrower;
use App\Models\LoanApplication;
use App\Models\LoanProduct;
use App\Services\Customer\BorrowerPortalService;
use Illuminate\Http\RedirectResponse;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Str;
use Illuminate\Validation\ValidationException;

final class ApplicationController extends Controller
{
    /** Statuses that mean "still being processed". */
    public const OPEN_STATUSES = ['submitted', 'pending', 'approved_pending_checker', 'approved'];

    public function __construct(private readonly BorrowerPortalService $portal) {}

    public function create(): \Illuminate\Contracts\View\View|RedirectResponse
    {
        $borrower = auth()->user()->borrower;

        if (! $borrower instanceof Borrower) {
            return redirect()->route('borrower.dashboard');
        }

        $latest = LoanApplication::query()
            ->where('borrower_id', $borrower->id)
            ->latest('id')
            ->first();

        return view('borrower.apply', [
            'borrower' => $borrower,
            'products' => LoanProduct::query()
                ->where('is_active', true)
                ->with(['interestRules', 'feeMatrices'])
                ->orderBy('min_principal')
                ->get(),
            'openApplication' => $latest !== null && in_array($latest->status, self::OPEN_STATUSES, true) ? $latest : null,
            'lastRejected' => $latest !== null && $latest->status === 'rejected' ? $latest : null,
            'hasActiveLoan' => $this->portal->currentLoan($borrower) !== null,
        ]);
    }

    public function store(StoreLoanApplicationRequest $request): RedirectResponse
    {
        $borrower = auth()->user()->borrower;

        if (! $borrower instanceof Borrower) {
            return redirect()->route('borrower.dashboard');
        }

        $product = LoanProduct::query()
            ->where('id', $request->integer('loan_product_id'))
            ->where('is_active', true)
            ->first();

        if ($product === null) {
            throw ValidationException::withMessages(['loan_product_id' => 'That loan product is not available.']);
        }

        $amount = bcadd((string) $request->validated('amount'), '0', 2);

        if (bccomp($amount, (string) $product->min_principal, 2) < 0 || bccomp($amount, (string) $product->max_principal, 2) > 0) {
            throw ValidationException::withMessages([
                'amount' => sprintf(
                    'For %s, choose between KES %s and KES %s.',
                    $product->name,
                    number_format((float) $product->min_principal, 0),
                    number_format((float) $product->max_principal, 0)
                ),
            ]);
        }

        DB::transaction(function () use ($borrower, $product, $amount, $request): void {
            // Lock the borrower row so a double tap cannot create two applications.
            Borrower::query()->where('id', $borrower->id)->lockForUpdate()->firstOrFail();

            $open = LoanApplication::query()
                ->where('borrower_id', $borrower->id)
                ->whereIn('status', self::OPEN_STATUSES)
                ->exists();

            if ($open) {
                throw ValidationException::withMessages(['amount' => 'You already have an application in progress.']);
            }

            if ($this->portal->currentLoan($borrower) !== null) {
                throw ValidationException::withMessages(['amount' => 'Please finish repaying your current loan before applying again.']);
            }

            $application = LoanApplication::create([
                'application_number' => 'APP-' . now()->format('ymd') . '-' . strtoupper(Str::random(6)),
                'borrower_id' => $borrower->id,
                'loan_product_id' => $product->id,
                'requested_principal' => $amount,
                'requested_tenure_days' => (int) $product->default_tenure_days,
                'status' => 'pending',
            ]);

            AuditTrail::create([
                'user_id' => auth()->id(),
                'action' => 'LOAN_APPLICATION_SUBMITTED',
                'auditable_type' => LoanApplication::class,
                'auditable_id' => $application->id,
                'old_values' => null,
                'new_values' => [
                    'product' => $product->product_code,
                    'amount' => $amount,
                    'tenure_days' => (int) $product->default_tenure_days,
                ],
                'ip_address' => substr((string) ($request->ip() ?? '127.0.0.1'), 0, 45),
                'user_agent' => substr((string) ($request->userAgent() ?? 'unknown'), 0, 255),
            ]);
        });

        return redirect()->route('borrower.apply.create')
            ->with('status', 'Application received. We will review it and update you here.');
    }
