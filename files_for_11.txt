<?php

declare(strict_types=1);

namespace App\Services\Underwriting;

use App\Enums\RiskBand;
use App\Models\Borrower;
use App\Models\BorrowerFinancialProfile;
use App\Models\CrbInquiryRecord;
use App\Models\CreditRuleEvaluation;
use App\Models\LoanApplication;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Log;

final class CreditUnderwritingEngine
{
    /**
     * Evaluates a loan application against statutory and risk underwriting rules.
     * Assigns risk bands, calculates credit limits, and records detailed rule logs.
     *
     * @param int $applicationId
     * @return array Summary of underwriting decision and assigned terms
     */
    public function evaluateApplication(int $applicationId): array
    {
        return DB::transaction(function () use ($applicationId) {
            /** @var LoanApplication $application */
            $application = LoanApplication::where('id', $applicationId)
                ->lockForUpdate()
                ->firstOrFail();

            /** @var Borrower $borrower */
            $borrower = Borrower::where('id', $application->borrower_id)
                ->lockForUpdate()
                ->firstOrFail();

            /** @var BorrowerFinancialProfile|null $profile */
            $profile = BorrowerFinancialProfile::where('borrower_id', $borrower->id)->first();

            /** @var CrbInquiryRecord|null $latestCrb */
            $latestCrb = CrbInquiryRecord::where('borrower_id', $borrower->id)
                ->orderByDesc('id')
                ->first();

            $passedCount = 0;
            $failedCount = 0;

            // 1. Hard Rule: Borrower Blacklist Inspection
            $isBlacklisted = $borrower->is_blacklisted;
            $this->recordRuleEvaluation(
                $application->id,
                'RULE_01_BLACKLIST_CHECK',
                'Borrower Fraud & Blacklist Inspection',
                ! $isBlacklisted ? 'PASSED' : 'FAILED',
                $isBlacklisted ? 'Blacklisted' : 'Clean',
                'Clean'
            );
            $isBlacklisted ? $failedCount++ : $passedCount++;

            // 2. Hard Rule: CRB Defaulted Accounts Audit
            $defaultedAccounts = $latestCrb ? $latestCrb->defaulted_accounts_count : 0;
            $crbPassed = $defaultedAccounts === 0;
            $this->recordRuleEvaluation(
                $application->id,
                'RULE_02_CRB_DEFAULTS',
                'CRB Active NPL/Defaulted Accounts Check',
                $crbPassed ? 'PASSED' : 'FAILED',
                (string) $defaultedAccounts,
                '0'
            );
            $crbPassed ? $passedCount++ : $failedCount++;

            // 3. Hard Rule: Debt-To-Income (DTI) Ratio Guard (Max 50.00%)
            $monthlyIncome = $profile ? (string) $profile->monthly_income : '0.00';
            $monthlyExpenses = $profile ? (string) $profile->monthly_expenses : '0.00';
            $existingDebts = $profile ? (string) $profile->existing_debt_obligations : '0.00';

            $totalObligations = bcadd($monthlyExpenses, $existingDebts, 2);
            $dtiRatio = bccomp($monthlyIncome, '0.00', 2) > 0
                ? bcdiv($totalObligations, $monthlyIncome, 4)
                : '1.0000';

            $dtiPassed = bccomp($dtiRatio, '0.5000', 4) <= 0;
            $dtiPercentage = bcmul($dtiRatio, '100', 2).'%';

            $this->recordRuleEvaluation(
                $application->id,
                'RULE_03_DTI_RATIO',
                'Debt-To-Income Ratio Below 50%',
                $dtiPassed ? 'PASSED' : 'FAILED',
                $dtiPercentage,
                '<= 50.00%'
            );
            $dtiPassed ? $passedCount++ : $failedCount++;

            // 4. Hard Rule: M-Pesa Cash Flow Capacity Test (Min 1.5x Requested Principal)
            $mpesaInflow = $profile ? (string) $profile->mpesa_average_monthly_inflow : '0.00';
            $requestedPrincipal = (string) $application->requested_principal;

            $requiredInflow = bcmul($requestedPrincipal, '1.50', 2);
            $inflowPassed = bccomp($mpesaInflow, $requiredInflow, 2) >= 0;

            $this->recordRuleEvaluation(
                $application->id,
                'RULE_04_MPESA_INFLOW_CAPACITY',
                'M-Pesa Average Monthly Inflow Capacity',
                $inflowPassed ? 'PASSED' : 'FAILED',
                "KES {$mpesaInflow}",
                ">= KES {$requiredInflow}"
            );
            $inflowPassed ? $passedCount++ : $failedCount++;

            // Decision Matrix & Term Assignment
            if ($failedCount > 0) {
                $status = 'rejected';
                $riskBand = RiskBand::D;
                $approvedLimit = '0.00';
                $rejectionReason = "Failed {$failedCount} underwriting risk evaluation rules.";
            } else {
                $status = 'approved_pending_checker';
                $rejectionReason = null;

                // Score-Based Risk Band Assignment
                $crbScore = $latestCrb ? $latestCrb->credit_score : 600;

                if ($crbScore >= 750) {
                    $riskBand = RiskBand::A;
                    $approvedLimit = bcmul($requestedPrincipal, '1.25', 2);
                } elseif ($crbScore >= 650) {
                    $riskBand = RiskBand::B;
                    $approvedLimit = $requestedPrincipal;
                } else {
                    $riskBand = RiskBand::C;
                    $approvedLimit = bcmul($requestedPrincipal, '0.75', 2);
                }
            }

            // Update Application state
            $application->update([
                'status' => $status,
                'rejection_reason' => $rejectionReason,
            ]);

            // Update Borrower risk profile and credit limit
            $borrower->update([
                'risk_band' => $riskBand,
                'credit_limit' => $approvedLimit,
            ]);

            Log::channel('fintech')->info("Underwriting complete for Application #{$application->id}: Status = {$status}, Risk Band = {$riskBand->value}, Approved Limit = KES {$approvedLimit}");

            return [
                'application_id' => $application->id,
                'status' => $status,
                'risk_band' => $riskBand->value,
                'approved_limit' => $approvedLimit,
                'rules_passed' => $passedCount,
                'rules_failed' => $failedCount,
            ];
        });
    }

    private function recordRuleEvaluation(
        int $applicationId,
        string $identifier,
        string $description,
        string $status,
        string $actualValue,
        string $thresholdValue
    ): void {
        CreditRuleEvaluation::create([
            'loan_application_id' => $applicationId,
            'rule_identifier' => $identifier,
            'rule_description' => $description,
            'status' => $status,
            'actual_value' => $actualValue,
            'threshold_value' => $thresholdValue,
        ]);
    }

final class B2cDisbursementService
{
    public function __construct(
        private readonly DarajaClient $darajaClient,
        private readonly SimSwapProtectionService $simSwapService,
        private readonly GeneralLedgerService $glService
    ) {}

    /**
     * Initiates an outbound M-Pesa B2C payment to the borrower's wallet.
     */
    public function initiateDisbursement(Loan $loan, ?CarbonImmutable $mockSimSwapTimestamp = null): MpesaB2cDisbursement
    {
        return DB::transaction(function () use ($loan, $mockSimSwapTimestamp): MpesaB2cDisbursement {
            $lockedLoan = Loan::query()->where('id', $loan->id)->lockForUpdate()->firstOrFail();
            $phoneNumber = (string) $lockedLoan->borrower->phone_number;

            // 1. Guard against SIM-Swap Fraud
            $this->simSwapService->assertPhoneIsSafeForDisbursement($phoneNumber, $mockSimSwapTimestamp);

            // 2. Prepare Daraja B2C Payload
            $token = $this->darajaClient->getAccessToken();
            $url = $this->darajaClient->getBaseUrl() . '/mpesa/b2c/v1/paymentrequest';
            $originatorConversationId = 'B2C-' . strtoupper(Str::random(12));

            $payload = [
                'InitiatorName'          => (string) config('mpesa.initiator_name', 'testapi'),
                'SecurityCredential'     => (string) config('mpesa.security_credential', 'testcredential'),
                'CommandID'              => 'BusinessPayment',
                'Amount'                 => (int) round((float) $lockedLoan->principal_disbursed),
                'PartyA'                 => (string) config('mpesa.b2c_shortcode', '600000'),
                'PartyB'                 => $phoneNumber,
                'Remarks'                => 'Loan Disbursement ' . $lockedLoan->loan_number,
                'QueueTimeOutURL'        => url('/api/v1/mpesa/b2c-timeout'),
                'ResultURL'              => url('/api/v1/mpesa/b2c-result'),
                'Occasion'               => 'Disbursement',
                'OriginatorConversationID' => $originatorConversationId,
            ];

            $response = Http::withToken($token)
                ->timeout((int) config('mpesa.http_timeout_seconds', 15))
                ->post($url, $payload);

            $responseData = $response->json() ?? [];
            $conversationId = (string) ($responseData['ConversationID'] ?? '');
            $responseCode = (string) ($responseData['ResponseCode'] ?? $response->status());
            $responseDescription = (string) ($responseData['ResponseDescription'] ?? $response->body());

            // 3. Persist initial B2C disbursement record
            $disbursement = MpesaB2cDisbursement::create([
                'loan_id'                    => $lockedLoan->id,
                'originator_conversation_id' => $originatorConversationId,
                'conversation_id'            => $conversationId !== '' ? $conversationId : null,
                'transaction_id'             => null,
                'recipient_phone'            => $phoneNumber,
                'amount'                     => $lockedLoan->principal_disbursed,
                'status'                     => $response->successful() ? 'initiated' : 'failed',
                'response_code'              => $responseCode,
                'response_description'       => $responseDescription,
                'disbursed_at'               => null,
            ]);

            AuditTrail::create([
                'user_id'        => auth()->id(),
                'action'         => 'MPESA_B2C_DISBURSEMENT_INITIATED',
                'auditable_type' => MpesaB2cDisbursement::class,
                'auditable_id'   => $disbursement->id,
                'old_values'     => null,
                'new_values'     => [
                    'amount'      => (string) $lockedLoan->principal_disbursed,
                    'phone'       => $phoneNumber,
                    'origin_conv' => $originatorConversationId,
                ],
                'ip_address'     => request()->ip() ?? '127.0.0.1',
                'user_agent'     => request()->userAgent() ?? 'System/CLI',
            ]);

            return $disbursement;
        });
    }

    /**
     * Processes the asynchronous Result callback from Safaricom B2C.
     *
     * @param array<string, mixed> $callbackData
     */
    public function processCallback(array $callbackData): MpesaB2cDisbursement
    {
        return DB::transaction(function () use ($callbackData): MpesaB2cDisbursement {
            $result = $callbackData['Result'] ?? $callbackData;
            $conversationId = (string) ($result['ConversationID'] ?? '');
            $originatorConversationId = (string) ($result['OriginatorConversationID'] ?? '');
            $resultCode = (int) ($result['ResultCode'] ?? -1);
            $resultDesc = (string) ($result['ResultDesc'] ?? 'No description');
            $transactionId = (string) ($result['TransactionID'] ?? 'TXN-' . Str::random(10));

            $disbursement = MpesaB2cDisbursement::query()
                ->where('originator_conversation_id', $originatorConversationId)
                ->orWhere('conversation_id', $conversationId)
                ->lockForUpdate()
                ->firstOrFail();

            $loan = Loan::query()->where('id', $disbursement->loan_id)->lockForUpdate()->firstOrFail();
            $now = CarbonImmutable::now();

            if ($resultCode === 0) {
                // SUCCESS: Funds delivered to borrower's phone wallet
                $disbursement->update([
                    'status'               => 'successful',
                    'transaction_id'       => $transactionId,
                    'response_code'        => (string) $resultCode,
                    'response_description' => $resultDesc,
                    'disbursed_at'         => $now,
                ]);

                $loan->update([
                    'status'       => LoanStatus::DISBURSED,
                    'disbursed_at' => $now,
                ]);

                // Create initial immutable debit in Loan Ledger
                LoanLedgerEntry::create([
                    'loan_id'                   => $loan->id,
                    'entry_type'                => 'disbursement',
                    'debit'                     => $loan->principal_disbursed,
                    'credit'                    => '0.00',
                    'running_principal_balance' => $loan->principal_disbursed,
                    'running_interest_balance'  => '0.00',
                    'running_fees_balance'      => '0.00',
                    'total_running_balance'     => $loan->principal_disbursed,
                    'reference'                 => $transactionId,
                    'notes'                     => 'Disbursement confirmed via M-Pesa B2C Receipt ' . $transactionId,
                ]);

                // Double-Entry GL Posting: Debit Loans Receivable, Credit M-Pesa Settlement
                $this->glService->recordLoanDisbursementPosting($loan, (string) $loan->principal_disbursed);

                // Cryptographically chained Outbox Event
                $prevEvent = OutboxEvent::query()->latest('id')->first();
                $prevHash = $prevEvent ? $prevEvent->current_event_hash : str_repeat('0', 64);

                $payload = [
                    'loan_id'         => $loan->id,
                    'loan_number'     => $loan->loan_number,
                    'disbursement_id' => $disbursement->id,
                    'transaction_id'  => $transactionId,
                    'amount'          => (string) $loan->principal_disbursed,
                    'phone'           => $disbursement->recipient_phone,
                    'disbursed_at'    => $now->toIso8601String(),
                ];

                $currentHash = hash('sha256', $prevHash . 'loan.disbursed' . json_encode($payload));

                OutboxEvent::create([
                    'aggregate_type'      => Loan::class,
                    'aggregate_id'        => $loan->id,
                    'event_name'          => 'loan.disbursed',
                    'payload'             => $payload,
                    'previous_event_hash' => $prevHash,
                    'current_event_hash'  => $currentHash,
                    'is_published'        => false,
                    'retry_count'         => 0,
                ]);
            } else {
                // FAILURE from Safaricom (e.g. Insufficient Utility Balance)
                $disbursement->update([
                    'status'               => 'failed',
                    'response_code'        => (string) $resultCode,
                    'response_description' => $resultDesc,
                ]);
            }

            return $disbursement;
        });
    }

    $disbursement = MpesaB2cDisbursement::create([
        'loan_id'                    => $this->loan->id,
        'originator_conversation_id' => 'B2C-CALLBACK-TEST-01',
        'conversation_id'            => 'CONV-TEST-01',
        'recipient_phone'            => '254712345678',
        'amount'                     => '15000.00',
        'status'                     => 'initiated',
    ]);

    $callbackPayload = [
        'Result' => [
            'ResultType'               => 0,
            'ResultCode'               => 0,
            'ResultDesc'               => 'The service request is processed successfully.',
            'OriginatorConversationID' => 'B2C-CALLBACK-TEST-01',
            'ConversationID'           => 'CONV-TEST-01',
            'TransactionID'            => 'QHD7823HG1',
        ],
    ];

    $updated = $this->service->processCallback($callbackPayload);

    expect($updated->status)->toBe('successful')
        ->and($updated->transaction_id)->toBe('QHD7823HG1');

    // Assert Loan status transitioned to DISBURSED
    $this->loan->refresh();
    expect($this->loan->status)->toBe(LoanStatus::DISBURSED)
        ->and($this->loan->disbursed_at)->not->toBeNull();

    // Assert initial debit in Loan Ledger
    $ledger = LoanLedgerEntry::where('loan_id', $this->loan->id)->first();
    expect($ledger)->not->toBeNull()
        ->and($ledger->entry_type)->toBe('disbursement')
        ->and(number_format((float) $ledger->running_principal_balance, 2, '.', ''))->toBe('15000.00');

    // Assert Outbox Event committed
    $outbox = OutboxEvent::where('aggregate_id', $this->loan->id)
        ->where('event_name', 'loan.disbursed')
        ->first();

    expect($outbox)->not->toBeNull()
        ->and($outbox->payload['transaction_id'])->toBe('QHD7823HG1');
});
