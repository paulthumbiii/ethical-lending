declare(strict_types=1);

namespace App\Services\Governance;

use App\Models\AuditTrail;
use App\Models\LoanApplication;
use App\Models\MakerCheckerRequest;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Log;
use RuntimeException;

final class MakerCheckerService
{
    /**
     * Creates a pending dual-control authorization request.
     *
     * @param int $makerUserId
     * @string $actionType (e.g., 'APPROVE_LOAN_DISBURSEMENT', 'WAIVE_INTEREST')
     * @string $targetEntityType
     * @int $targetEntityId
     * @array $payload
     * @return MakerCheckerRequest
     */
    public function createPendingRequest(
        int $makerUserId,
        string $actionType,
        string $targetEntityType,
        int $targetEntityId,
        array $payload = []
    ): MakerCheckerRequest {
        return MakerCheckerRequest::create([
            'maker_user_id' => $makerUserId,
            'action_type' => $actionType,
            'target_entity_type' => $targetEntityType,
            'target_entity_id' => $targetEntityId,
            'payload' => $payload,
            'status' => 'pending',
        ]);
    }

    /**
     * Authorizes and executes a pending request under strict non-self-approval enforcement.
     *
     * @param int $requestId
     * @param int $checkerUserId
     * @param string|null $checkerNotes
     * @return array
     * @throws RuntimeException
     */
    public function authorizeRequest(int $requestId, int $checkerUserId, ?string $checkerNotes = null): array
    {
        return DB::transaction(function () use ($requestId, $checkerUserId, $checkerNotes) {
            /** @var MakerCheckerRequest $request */
            $request = MakerCheckerRequest::where('id', $requestId)
                ->lockForUpdate()
                ->firstOrFail();

            if ($request->status !== 'pending') {
                throw new RuntimeException("Maker-Checker Request #{$requestId} is not in pending status.");
            }

            // Dual-Control Requirement: Checker MUST NOT be the Maker.
            // Compared as integers so a driver that returns ids as strings cannot slip past the check.
            if ((int) $request->maker_user_id === $checkerUserId) {
                Log::channel('fintech')->warning("Dual-Control Violation Attempt: User #{$checkerUserId} attempted self-approval on Request #{$requestId}");
                throw new RuntimeException("Dual-Control Violation: You cannot authorize a request you initiated.");
            }

            $request->update([
                'status' => 'approved',
                'checker_user_id' => $checkerUserId,
                'checker_notes' => $checkerNotes,
                'actioned_at' => now(),
            ]);

            // Execute specific state mutation based on action type
            if ($request->action_type === 'APPROVE_LOAN_APPLICATION') {
                /** @var LoanApplication|null $app */
                $app = LoanApplication::where('id', $request->target_entity_id)->first();
                if ($app) {
                    $app->update(['status' => 'approved']);
                }
            }

            // Record immutable Audit Trail
            AuditTrail::create([
                'user_id' => $checkerUserId,
                'action' => 'MAKER_CHECKER_AUTHORIZED',
                'auditable_type' => MakerCheckerRequest::class,
                'auditable_id' => $request->id,
                'old_values' => ['status' => 'pending'],
                'new_values' => [
                    'status' => 'approved',
                    'action_type' => $request->action_type,
                    'checker_user_id' => $checkerUserId,
                ],
                'ip_address' => request()->ip() ?? '127.0.0.1',
                'user_agent' => request()->userAgent() ?? 'SYSTEM',
            ]);

            Log::channel('fintech')->info("Maker-Checker Request #{$requestId} [{$request->action_type}] authorized by Checker #{$checkerUserId}");

            return [
                'request_id' => $request->id,
                'action_type' => $request->action_type,
                'status' => 'approved',
            ];
        });
    }

    /**
     * Rejects a pending dual-control request and records the decision on the audit trail.
     */
    public function rejectRequest(int $requestId, int $checkerUserId, string $reason): array
    {
        return DB::transaction(function () use ($requestId, $checkerUserId, $reason) {
            /** @var MakerCheckerRequest $request */
            $request = MakerCheckerRequest::where('id', $requestId)
                ->lockForUpdate()
                ->firstOrFail();

            if ($request->status !== 'pending') {
                throw new RuntimeException("Maker-Checker Request #{$requestId} is not in pending status.");
            }

            if ((int) $request->maker_user_id === $checkerUserId) {
                Log::channel('fintech')->warning("Dual-Control Violation Attempt: User #{$checkerUserId} attempted to reject own Request #{$requestId}");
                throw new RuntimeException("Dual-Control Violation: You cannot reject/action a request you initiated.");
            }

            $request->update([
                'status' => 'rejected',
                'checker_user_id' => $checkerUserId,
                'checker_notes' => $reason,
                'actioned_at' => now(),
            ]);

            AuditTrail::create([
                'user_id' => $checkerUserId,
                'action' => 'MAKER_CHECKER_REJECTED',
                'auditable_type' => MakerCheckerRequest::class,
                'auditable_id' => $request->id,
                'old_values' => ['status' => 'pending'],
                'new_values' => [
                    'status' => 'rejected',
                    'action_type' => $request->action_type,
                    'checker_user_id' => $checkerUserId,
                    'reason' => $reason,
                ],
                'ip_address' => request()->ip() ?? '127.0.0.1',
                'user_agent' => request()->userAgent() ?? 'SYSTEM',
            ]);

            Log::channel('fintech')->info("Maker-Checker Request #{$requestId} [{$request->action_type}] rejected by Checker #{$checkerUserId}");

            return [
                'request_id' => $request->id,
                'status' => 'rejected',
            ];
        });
    }

use App\Http\Controllers\Controller;

use Illuminate\Http\RedirectResponse;
use RuntimeException;

final class MakerCheckerController extends Controller
{
    public function __construct(private readonly MakerCheckerService $service)
    {
    }

    public function index(): View
    {
        $pendingCount = MakerCheckerRequest::query()->where('status', 'pending')->count();

        $pending = MakerCheckerRequest::query()
            ->with('maker')
            ->where('status', 'pending')
            ->oldest('id')
            ->limit(50)
            ->get();

        $recent = MakerCheckerRequest::query()
            ->with(['maker', 'checker'])
            ->whereIn('status', ['approved', 'rejected'])
            ->latest('actioned_at')
            ->limit(10)
            ->get();

        return view('lender.maker-checker.index', [
            'pendingCount' => $pendingCount,
            'pending'      => $pending,
            'recent'       => $recent,
        ]);
    }

    public function approve(
        ReviewMakerCheckerRequest $request,
        MakerCheckerRequest $makerCheckerRequest
    ): RedirectResponse {
        try {
            $this->service->authorizeRequest(
                $makerCheckerRequest->id,
                (int) $request->user()->id,
                $request->validated('notes')
            );
        } catch (RuntimeException $e) {
            return back()->withErrors(['review' => $e->getMessage()]);
        }

        return back()->with('status', "Request #{$makerCheckerRequest->id} approved.");
    }

    public function reject(
        ReviewMakerCheckerRequest $request,
        MakerCheckerRequest $makerCheckerRequest
    ): RedirectResponse {
        try {
            $this->service->rejectRequest(
                $makerCheckerRequest->id,
                (int) $request->user()->id,
                (string) $request->validated('notes')
            );
        } catch (RuntimeException $e) {
            return back()->withErrors(['review' => $e->getMessage()]);
        }

        return back()->with('status', "Request #{$makerCheckerRequest->id} rejected.");
    }

        $waiting = LoanApplication::query()
            ->with(['borrower', 'product'])

            ->limit(100)
            ->get();

        $decided = LoanApplication::query()

            ->whereIn('status', ['approved_pending_checker', 'rejected', 'approved'])
            ->latest('id')
            ->limit(20)

            'decided' => $decided,

    public function show(LoanApplication $loanApplication): View
    {
        $loanApplication->load([
            'borrower.financialProfile',
            'product',
            'underwriter',
            'ruleEvaluations' => fn ($query) => $query->orderBy('id'),
        ]);

        $statements = MpesaStatementUpload::query()
            ->with('reviewer')
            ->where('loan_application_id', $loanApplication->id)
            ->latest('id')
            ->get();

        $today = now()->toDateString();

        $hasAcceptedStatement = $statements->contains(

                && $s->valid_until !== null
                && $s->valid_until->toDateString() >= $today
        );

        $checkable = in_array($loanApplication->status, ApplicationReviewService::CHECKABLE_STATUSES, true);
        $isUnderwriter = (bool) auth()->user()?->hasAnyRole(UserRole::underwriters());

        return view('lender.applications.show', [
            'application' => $loanApplication,
            'statements' => $statements,
            'hasAcceptedStatement' => $hasAcceptedStatement,
            'isUnderwriter' => $isUnderwriter,
            'checkable' => $checkable,
            'canAct' => $checkable && $isUnderwriter,
        ]);
    }

        UpdateApplicationInflowRequest $request,

namespace App\Services\Underwriting;

use App\Enums\StatementReviewStatus;

use App\Models\LoanApplication;
use App\Models\MpesaStatementUpload;
use Illuminate\Support\Facades\DB;
use RuntimeException;

final class ApplicationReviewService
{
    /** Statuses that still wait for a credit check. */
    public const CHECKABLE_STATUSES = ['submitted', 'pending'];

    public function __construct(private readonly CreditUnderwritingEngine $engine)
    {
    }

    /**
     * Staff-verified M-Pesa average monthly inflow. Audited with old and new value.
     */
    public function updateInflow(
        int $applicationId,
        int $userId,
        string $newInflow,
        string $ip,
        string $userAgent
    ): void {
        $newInflow = bcadd($newInflow, '0', 2);

        DB::transaction(function () use ($applicationId, $userId, $newInflow, $ip, $userAgent): void {
            $application = LoanApplication::query()
                ->where('id', $applicationId)
                ->lockForUpdate()
                ->firstOrFail();

            $this->assertCheckable($application);

            $profile = BorrowerFinancialProfile::query()
                ->where('borrower_id', $application->borrower_id)
                ->lockForUpdate()
                ->first();

            $old = $profile !== null
                ? bcadd((string) $profile->mpesa_average_monthly_inflow, '0', 2)
                : null;

            if ($profile === null) {
                BorrowerFinancialProfile::create([
                    'borrower_id' => $application->borrower_id,
                    'monthly_income' => '0.00',
                    'monthly_expenses' => '0.00',
                    'existing_debt_obligations' => '0.00',
                    'mpesa_average_monthly_inflow' => $newInflow,
                ]);
            } else {
                $profile->update(['mpesa_average_monthly_inflow' => $newInflow]);
            }

            AuditTrail::create([
                'user_id' => $userId,
                'action' => 'APPLICATION_INFLOW_UPDATED',
                'auditable_type' => LoanApplication::class,
                'auditable_id' => $application->id,
                'old_values' => ['mpesa_average_monthly_inflow' => $old],
                'new_values' => ['mpesa_average_monthly_inflow' => $newInflow],
                'ip_address' => substr($ip, 0, 45),
                'user_agent' => substr($userAgent, 0, 255),
            ]);
        });
    }

    /**
     * Runs the underwriting engine once on a pending application,
     * and only when an accepted, in-date statement backs it.
     *
     * @return array<string, mixed>
     */
    public function runCheck(int $applicationId, int $userId, string $ip, string $userAgent): array
    {
        return DB::transaction(function () use ($applicationId, $userId, $ip, $userAgent): array {
            $application = LoanApplication::query()
                ->where('id', $applicationId)
                ->lockForUpdate()
                ->firstOrFail();

            $this->assertCheckable($application);

            $hasValidStatement = MpesaStatementUpload::query()
                ->where('loan_application_id', $application->id)
                ->where('review_status', StatementReviewStatus::Accepted->value)
                ->whereDate('valid_until', '>=', now()->toDateString())
                ->exists();

            if (! $hasValidStatement) {
                throw new RuntimeException('Accept a valid M-Pesa statement before running the credit check.');
            }

            $oldStatus = (string) $application->status;

            // Page always shows the latest run only.
            CreditRuleEvaluation::query()
                ->where('loan_application_id', $application->id)
                ->delete();

            $result = $this->engine->evaluateApplication($application->id);

            LoanApplication::query()
                ->whereKey($application->id)
                ->update(['underwriter_id' => $userId]);

            AuditTrail::create([
                'user_id' => $userId,
                'action' => 'APPLICATION_CREDIT_CHECK_RUN',
                'auditable_type' => LoanApplication::class,
                'auditable_id' => $application->id,
                'old_values' => ['status' => $oldStatus],
                'new_values' => [
                    'status' => $result['status'],
                    'risk_band' => $result['risk_band'],
                    'approved_limit' => $result['approved_limit'],
                    'rules_passed' => $result['rules_passed'],
                    'rules_failed' => $result['rules_failed'],
                ],
                'ip_address' => substr($ip, 0, 45),
                'user_agent' => substr($userAgent, 0, 255),
            ]);

            return $result;
        });
    }

    private function assertCheckable(LoanApplication $application): void
    {
        if (! in_array($application->status, self::CHECKABLE_STATUSES, true)) {
            throw new RuntimeException('This application has already been checked.');
        }
    }

        'name'         => 'Test ' . $role->value,
        'email'        => $role->value . '.' . $phone . '@ethical-lending.test',
        'password'     => bcrypt('Demo@12345'),
        'phone_number' => $phone,
        'is_active'    => true,
    ]);

    $user->assignRole($role->value);

    return $user;
}

function mcdRequest(User $maker, string $action = 'WAIVE_INTEREST'): MakerCheckerRequest
{
    return MakerCheckerRequest::create([
        'maker_user_id'      => $maker->id,
        'action_type'        => $action,
        'target_entity_type' => 'App\\Models\\Loan',
        'target_entity_id'   => 1,

        'status'             => 'pending',
    ]);
