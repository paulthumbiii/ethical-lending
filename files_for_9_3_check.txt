<?php

declare(strict_types=1);

namespace App\Services\Customer;

use App\Enums\LoanStatus;
use App\Models\Borrower;
use App\Models\Loan;
use App\Models\LoanLedgerEntry;
use App\Services\Servicing\RestructuringEngine;
use Carbon\CarbonImmutable;
use Illuminate\Support\Facades\DB;

final class BorrowerPortalService
{
    /** Extra days of time a borrower can choose from. */
    public const EXTENSION_OPTIONS = [14, 21, 30, 45, 60];

    public const DEFAULT_EXTENSION_DAYS = 30;

    /** Mirrors the threshold inside RestructuringEngine. The engine remains the final judge. */
    private const MINIMUM_ELIGIBILITY_RATIO = '0.30';

    public function __construct(private readonly RestructuringEngine $engine)
    {
    }

    /**
     * Everything the borrower dashboard needs.
     *
     * @return array<string, mixed>
     */
    public function summary(Borrower $borrower): array
    {
        $loan = $this->currentLoan($borrower);

        if ($loan === null) {
            return ['loan' => null];
        }

        $balances = $this->balances($loan);
        $owed = $this->owed($balances);
        $due = $loan->due_date !== null ? CarbonImmutable::parse($loan->due_date) : null;

        return [
            'loan'       => $loan,
            'balances'   => $balances,
            'owed'       => $owed,
            'due_date'   => $due?->format('j M Y'),
            'is_overdue' => $due !== null && $due->endOfDay()->isPast() && bccomp($owed, '0.00', 2) > 0,
            'meter'      => $this->inDuplumMeter($loan),
            'progress'   => $this->progress($loan, $owed),
            'relief'     => $this->canRequestRelief($loan) ? $this->reliefPreview($borrower, $balances, $loan) : null,
        ];
    }

    /** The borrower's latest loan that is not settled or written off. */
    public function currentLoan(Borrower $borrower): ?Loan
    {
        /** @var Loan|null $loan */
        $loan = $borrower->loans()
            ->whereNotIn('status', [LoanStatus::SETTLED->value, LoanStatus::WRITTEN_OFF->value])
            ->with('inDuplumTracking')
            ->latest('id')
            ->first();

        return $loan;
    }

    /**
     * Latest balances from the append-only ledger.
     *
     * @return array{principal: string, interest: string, fees: string}
     */
    public function balances(Loan $loan): array
    {
        $latest = LoanLedgerEntry::query()->where('loan_id', $loan->id)->latest('id')->first();

        if ($latest === null) {
            return [
                'principal' => (string) $loan->principal_disbursed,
                'interest'  => '0.00',
                'fees'      => '0.00',
            ];
        }

        return [
            'principal' => (string) $latest->running_principal_balance,
            'interest'  => (string) $latest->running_interest_balance,
            'fees'      => (string) $latest->running_fees_balance,
        ];
    }

    /**
     * Data for the Section 44A safety meter.
     * States: not_triggered (no default yet), accruing (charges counting), frozen (cap reached).
     *
     * @return array{state: string, accrued: string, cap: string, percent: string, headroom: string}
     */
    public function inDuplumMeter(Loan $loan): array
    {
        $tracking = $loan->inDuplumTracking;

        if ($tracking === null) {
            return [
                'state'    => 'not_triggered',
                'accrued'  => '0.00',
                'cap'      => '0.00',
                'percent'  => '0.00',
                'headroom' => '0.00',
            ];
        }

        $accrued = bcadd(
            (string) $tracking->total_post_default_interest,
            (string) $tracking->total_post_default_penalties,
            2
        );
        $cap = (string) $tracking->statutory_cap_limit;

        $percent = '0.00';
        if (bccomp($cap, '0.00', 2) > 0) {
            $percent = bcmul(bcdiv($accrued, $cap, 4), '100', 2);
            if (bccomp($percent, '100.00', 2) > 0) {
                $percent = '100.00';
            }
        }

        $headroom = bcsub($cap, $accrued, 2);
        if (bccomp($headroom, '0.00', 2) < 0) {
            $headroom = '0.00';
        }

        $frozen = (bool) $tracking->is_frozen || (bool) $loan->is_in_duplum_locked;

        return [
            'state'    => $frozen ? 'frozen' : 'accruing',
            'accrued'  => $accrued,
            'cap'      => $cap,
            'percent'  => $percent,
            'headroom' => $headroom,
        ];
    }

    /** Relief is offered on defaulted loans, or disbursed loans past their due date. */
    public function canRequestRelief(Loan $loan): bool
    {
        if ($loan->status === LoanStatus::DEFAULTED) {
            return true;
        }

        return $loan->status === LoanStatus::DISBURSED
            && $loan->due_date !== null
            && CarbonImmutable::parse($loan->due_date)->endOfDay()->isPast();
    }

    /**
     * A read-only preview that follows the same rules as RestructuringEngine
     * (penalties forgiven, interest and the processing fee kept, balance split into two instalments).
     * Pass the loan so the kept fee can be read from the ledger. Nothing is saved here.
     *
     * @param  array{principal: string, interest: string, fees: string}  $balances
     * @return array<string, mixed>
     */
    public function reliefPreview(Borrower $borrower, array $balances, ?Loan $loan = null): array
    {
        $ratio = $this->engine->calculateRepaymentRatio($borrower);

        $owedToday = $this->owed($balances);
        $kept = $loan !== null ? $this->keptFee($loan->id, $balances['fees']) : '0.00';
        $waived = bcsub($balances['fees'], $kept, 2);
        $owedAfter = bcadd(bcadd($balances['principal'], $balances['interest'], 2), $kept, 2);

        $first = bcdiv($owedAfter, '2.00', 2);
        $remainder = bcsub($owedAfter, bcmul($first, '2.00', 2), 2);
        $second = bcadd($first, $remainder, 2);

        $today = CarbonImmutable::now();
        $schedule = [];
        foreach (self::EXTENSION_OPTIONS as $days) {
            $schedule[$days] = [
                'first_due'  => $today->addDays(intdiv($days, 2))->format('j M Y'),
                'second_due' => $today->addDays($days)->format('j M Y'),
            ];
        }

        return [
            'eligible'      => bccomp($ratio, self::MINIMUM_ELIGIBILITY_RATIO, 2) >= 0,
            'owed_today'    => $owedToday,
            'waived'        => $waived,
            'kept_fee'      => $kept,
            'owed_after'    => $owedAfter,
            'first_amount'  => $first,
            'second_amount' => $second,
            'options'       => self::EXTENSION_OPTIONS,
            'default_days'  => self::DEFAULT_EXTENSION_DAYS,
            'schedule'      => $schedule,
        ];
    }

    /**
     * Creates and accepts a restructuring proposal in one transaction,
     * using the same engine the lender side relies on.
     */
    public function acceptRelief(Loan $loan, int $extensionDays): Loan
    {
        if (! in_array($extensionDays, self::EXTENSION_OPTIONS, true)) {
            throw new \InvalidArgumentException('Unsupported extension period.');
        }

        if (! $this->canRequestRelief($loan)) {
            throw new \DomainException('This loan is not in arrears, so relief cannot be requested.');
        }

        return DB::transaction(function () use ($loan, $extensionDays): Loan {
            $proposal = $this->engine->generateProposal($loan, $extensionDays);

            return $this->engine->acceptProposal($proposal);
        });
    }

    /** @param array{principal: string, interest: string, fees: string} $balances */
    private function owed(array $balances): string
    {
        return bcadd(bcadd($balances['principal'], $balances['interest'], 2), $balances['fees'], 2);
    }

    /**
     * Repayment progress for the loan card. Repaid is what the payment waterfall allocated to
     * fees, interest and principal (overpayment credit is not counted). The total is repaid plus
     * what is still owed. Display only; BCMath on strings, no floats. Percent is a whole number 0 to 100.
     *
     * @return array{repaid: string, total: string, percent: string}
     */
    private function progress(Loan $loan, string $owed): array
    {
        $repaid = '0.00';

        foreach ($loan->waterfallLogs()->get() as $log) {
            $part = bcadd(
                bcadd((string) ($log->allocated_to_fees ?? '0.00'), (string) ($log->allocated_to_interest ?? '0.00'), 2),
                (string) ($log->allocated_to_principal ?? '0.00'),
                2
            );
            $repaid = bcadd($repaid, $part, 2);
        }

        $total = bcadd($repaid, $owed, 2);
        $percent = '0';

        if (bccomp($total, '0.00', 2) > 0) {
            $percent = bcdiv(bcmul($repaid, '100', 2), $total, 0);

            if (bccomp($percent, '100', 0) > 0) {
                $percent = '100';
            }
        }

        return ['repaid' => $repaid, 'total' => $total, 'percent' => $percent];
    }

    /**
     * Same rule as RestructuringEngine::keptFee: the total ever charged as FEE_CHARGED,
     * capped at what is still owed in fees. If you change one, change both.
     */
    private function keptFee(int $loanId, string $feesOwed): string
    {
        $charged = '0.00';

        $rows = LoanLedgerEntry::query()
            ->where('loan_id', $loanId)
            ->where('entry_type', 'FEE_CHARGED')
            ->get();

        foreach ($rows as $row) {
            $charged = bcadd($charged, (string) $row->debit, 2);
        }

        return bccomp($charged, $feesOwed, 2) < 0 ? $charged : $feesOwed;
    }
}

<?php

declare(strict_types=1);

namespace App\Http\Controllers\Borrower;

use App\Enums\LoanStatus;
use App\Http\Controllers\Controller;
use App\Models\BorrowerKycDocument;
use App\Services\Customer\BorrowerPortalService;
use Carbon\CarbonImmutable;
use Illuminate\Http\Request;
use Illuminate\Support\Collection;
use Illuminate\View\View;

final class AccountController extends Controller
{
    public function __construct(private readonly BorrowerPortalService $portal)
    {
    }

    public function show(Request $request): View
    {
        $borrower = $request->user()?->borrower;
        $loans = [];

        /** @var Collection<string, BorrowerKycDocument> $documents */
        $documents = collect();

        if ($borrower !== null) {
            foreach ($borrower->loans()->latest('id')->get() as $loan) {
                $owed = '0.00';
                $open = $loan->status !== LoanStatus::SETTLED && $loan->status !== LoanStatus::WRITTEN_OFF;

                if ($open) {
                    $b = $this->portal->balances($loan);
                    $owed = bcadd(bcadd($b['principal'], $b['interest'], 2), $b['fees'], 2);
                }

                $schedule = [];
                // Instalments replaced by a relief plan are history, not something the borrower should pay.
                $items = $loan->amortizationSchedules()
                    ->where('status', '!=', 'superseded')
                    ->orderBy('installment_number')
                    ->get();

                foreach ($items as $item) {
                    $paid = bcadd(
                        bcadd((string) $item->fees_paid, (string) $item->interest_paid, 2),
                        (string) $item->principal_paid,
                        2
                    );
                    $left = bcsub((string) $item->total_installment_due, $paid, 2);
                    $left = bccomp($left, '0.00', 2) < 0 ? '0.00' : $left;
                    $status = (string) $item->status;

                    // "Overdue" is derived: there is no overdue status on an instalment.
                    $dueDay = CarbonImmutable::parse($item->due_date->toDateString(), 'Africa/Nairobi')->endOfDay();
                    $overdue = $open
                        && $status !== 'paid'
                        && bccomp($left, '0.00', 2) > 0
                        && $dueDay->isPast();

                    $schedule[] = [
                        'number'  => (int) $item->installment_number,
                        'due'     => $item->due_date,
                        'left'    => $left,
                        'status'  => $status,
                        'overdue' => $overdue,
                    ];
                }

                $loans[] = ['loan' => $loan, 'owed' => $owed, 'schedule' => $schedule];
            }

            $documents = $borrower->kycDocuments()->get()->keyBy('document_type');
        }

        return view('borrower.account', [
            'borrower'  => $borrower,
            'loans'     => $loans,
            'documents' => $documents,
        ]);
    }
}

Data protection (Kenya DPA 2019): informed consent recorded with IP, user agent and timestamp before processing a loan; M-Pesa statement consent is a required tick box written to the audit trail. ⚠️ No retention job exists for statement files or KYC ID files. ⚠️ data_protection_consents once showed 1 row; origin never checked (Phase 15).
Excise duty: 20% on processing/admin fees; the GL splits fee postings 80% revenue / 20% KRA excise payable. The code treats the 20% as PART of the fee (excise = fee × 0.20, net income = fee − excise). Kenyan excise on lender fees is normally charged ON TOP. ⚠️ The user was asked “part or on top?” several times and never answered. Default: “part of the fee”. If he later says “on top”, the change touches LoanFeeService, GeneralLedgerService::recordFeeWithExcisePosting, the contract generator and its view, the “(includes 20% excise duty)” strings (Apply page, disbursement panel, LoanFeeService ledger note) and their tests; the schedule follows automatically. The landing-page calculator (public/welcome/landing.js) also shows excise as “included” and would need updating.

Money rules: all money columns DECIMAL(12,2); no floats in calculations; BCMath (bcadd, bcsub, bcmul, bcdiv, bccomp) on strings. number_format((float) $x, 2) is for DISPLAY only. ⚠️ Known small deviation: FinanceController::money().

Key design choices from Phases 1-13 (keep respecting them)
Append-only ledger. Balances always come from the latest loan_ledger_entries row, never updated in place.

Auth. Hand-written login (email or phone plus password), no Breeze, single web guard. Inactive and role-less accounts are blocked. Auth audit rows come from the RecordAuthActivity listener.
Role gates in one place. UserRole helpers feed both routes and nav.

    $catalog = $products->map(function ($product): array {
        $rule = $product->interestRules->first();
        $fees = [];

        foreach ($product->feeMatrices as $fee) {
            $type = strtolower((string) $fee->calculation_type);

            if ($type === 'percentage') {
                $fees[] = [
                    'name'   => (string) $fee->fee_name,
                    'kind'   => 'pct',
                    'value'  => (string) $fee->fee_percentage,
                    'excise' => (bool) $fee->is_excise_duty_applicable,
                ];
            } elseif ($type === 'fixed' || $type === 'flat') {
                $fees[] = [

                    'kind'   => 'fixed',
                    'value'  => (string) $fee->fee_amount,
                    'excise' => (bool) $fee->is_excise_duty_applicable,
                ];
            }
        }

        return [
            'id'   => (string) $product->id,
            'min'  => (int) $product->min_principal,
            'max'  => (int) $product->max_principal,
            'days' => (int) $product->default_tenure_days,
            'rate' => $rule ? (string) $rule->annual_rate : null,
            'fees' => $fees,
        ];
    })->values()->all();

                document.addEventListener('alpine:init', () => {
                    Alpine.data('applyQuote', (catalog, startPid, startAmount) => ({
                        catalog,
                        pid: startPid,
                        amount: startAmount,

                        get product() {
                            return this.catalog.find(p => p.id === String(this.pid)) || this.catalog[0];
                        },
                        get principal() {
                            return this.amount === '' ? NaN : parseInt(this.amount, 10);
                        },
                        get valid() {
                            return Number.isInteger(this.principal) && this.principal >= this.product.min && this.principal <= this.product.max;
                        },
                        get principalCents() {
                            return this.principal * 100;
                        },
                        get sliderValue() {
                            if (! Number.isInteger(this.principal)) return this.product.min;
                            return Math.min(Math.max(this.principal, this.product.min), this.product.max);
                        },

                        // "18.00" -> 1800 (hundredths), so everything below is integer maths.
                        hundredths(text) {
                            const [whole, frac = ''] = String(text).split('.');
                            return parseInt(whole, 10) * 100 + parseInt((frac + '00').slice(0, 2), 10);
                        },
                        // Round half up for non-negative integers: round(num / den).
                        div(num, den) {
                            return Math.floor((2 * num + den) / (2 * den));
                        },

                        get quote() {
                            const p = this.product;
                            const cents = this.principalCents;
                            let interestCents = null;
                            let total = cents;

                            // principal x rate x days / 36,500, same as the contract.
                            if (p.rate !== null) {
                                interestCents = this.div(cents * this.hundredths(p.rate) * p.days, 36500 * 100);
                                total += interestCents;
                            }

                            const fees = [];
                            for (const f of p.fees) {
                                const feeCents = f.kind === 'pct'
                                    ? this.div(cents * this.hundredths(f.value), 10000)
                                    : this.hundredths(f.value);
                                if (feeCents <= 0) continue;
                                fees.push({ name: f.name, excise: f.excise, cents: feeCents });
                                total += feeCents;
                            }

                            return { interestCents, fees, totalCents: total };
                        },

                        fmt(cents) {
                            return 'KES ' + (cents / 100).toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
                        },
                        fmtWhole(cents) {
                            return 'KES ' + (cents / 100).toLocaleString('en-US', { maximumFractionDigits: 0 });
                        },
                    }));
                });
